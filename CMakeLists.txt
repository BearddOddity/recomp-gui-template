cmake_minimum_required(VERSION 3.16)
project(recomp_gui C)
set(CMAKE_C_STANDARD 11)

# Core (portable C11) + a renderer + the overlay for the platform.
#   Windows : Direct3D 11 renderer, overlay with XInput / performance counters, Present hook
#   macOS   : Metal renderer + portable overlay (host feeds input)
#   Linux   : OpenGL 3.3 / ES 3.0 renderer (loader supplied by the host) + portable overlay
if(WIN32)
    add_library(recomp_gui STATIC
        src/gui.c src/gui_d3d11.c src/gui_overlay.c src/gui_host.c src/gui_present_hook.c)
    target_link_libraries(recomp_gui PUBLIC d3d11 dxgi dxguid d3dcompiler pdh psapi xinput)
else()
    add_library(recomp_gui STATIC src/gui.c src/gui_overlay_portable.c src/gui_host.c)
    target_link_libraries(recomp_gui PUBLIC m)
    find_package(Threads REQUIRED)
    target_link_libraries(recomp_gui PUBLIC Threads::Threads)
    if(APPLE)
        enable_language(OBJC)
        target_sources(recomp_gui PRIVATE src/gui_metal.m)
        set_source_files_properties(src/gui_metal.m PROPERTIES COMPILE_FLAGS "-fobjc-arc")
        target_link_libraries(recomp_gui PUBLIC "-framework Metal" "-framework Foundation")
    else()
        target_sources(recomp_gui PRIVATE src/gui_gl.c)
    endif()
endif()

target_include_directories(recomp_gui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)

if(MSVC)
    target_compile_options(recomp_gui PRIVATE /W4 /O2)
else()
    target_compile_options(recomp_gui PRIVATE -Wall -Wextra -O2)
endif()

# Example menu (compiled, not linked into anything): proves it builds.
add_library(recomp_gui_example OBJECT examples/debug_menu.c)
target_link_libraries(recomp_gui_example PRIVATE recomp_gui)

add_executable(gui_smoke tests/gui_smoke.c)
target_link_libraries(gui_smoke PRIVATE recomp_gui)

if(WIN32)
    add_executable(present_hook_test tests/present_hook_test.c)
    target_link_libraries(present_hook_test PRIVATE recomp_gui)
else()
    add_executable(portable_overlay_test tests/portable_overlay_test.c)
    target_link_libraries(portable_overlay_test PRIVATE recomp_gui)
    if(APPLE)
        add_executable(metal_render_test tests/metal_render_test.m)
        set_source_files_properties(tests/metal_render_test.m PROPERTIES COMPILE_FLAGS "-fobjc-arc")
        target_link_libraries(metal_render_test PRIVATE recomp_gui)
    else()
        find_library(EGL_LIBRARY EGL)
        find_path(EGL_INCLUDE_DIR EGL/egl.h)
        if(EGL_LIBRARY AND EGL_INCLUDE_DIR)
            add_executable(gl_render_test tests/gl_render_test.c)
            target_include_directories(gl_render_test PRIVATE ${EGL_INCLUDE_DIR})
            target_link_libraries(gl_render_test PRIVATE recomp_gui ${EGL_LIBRARY})
        endif()
    endif()
endif()
